// Parity-only overlay. Calls the owning mount/disposal functions unchanged.
package hostmcp

import (
	"context"
	"encoding/json"
	"github.com/modelcontextprotocol/go-sdk/mcp"
	provider "github.com/wunderous/host-agents/contracts/provider"
	"github.com/wunderous/host-agents/internal/cordis"
	adapter "github.com/wunderous/host-agents/internal/cordis/mcp"
	"github.com/wunderous/host-agents/internal/tools"
	"strings"
)

func ParityProviderMounts(raw json.RawMessage) []map[string]any {
	var input struct {
		Steps []struct {
			Candidate  bool                      `json:"candidate"`
			Operation  provider.Operation        `json:"operation"`
			Action     string                    `json:"action"`
			Manifest   provider.InstallManifest  `json:"manifest"`
			Generation string                    `json:"generation"`
			Provider   string                    `json:"provider"`
			Service    string                    `json:"service"`
			Adapter    string                    `json:"adapter"`
			IDs        []string                  `json:"ids"`
			Families   []string                  `json:"families"`
			Restore    cordis.ProviderGeneration `json:"restore"`
		} `json:"steps"`
	}
	if err := json.Unmarshal(raw, &input); err != nil {
		panic(err)
	}
	server := &Server{providerContext: cordis.NewContext(), providerLifecycle: cordis.NewProviderLifecycleManager(cordis.DrainPolicy{}), providerCandidates: map[string]*adapter.Adapter{}}
	connections := map[string]*adapter.Adapter{}
	observations := []map[string]any{}
	for _, step := range input.Steps {
		var err error
		result := map[string]any{}
		switch step.Action {
		case "adapter":
			connections[step.Adapter] = &adapter.Adapter{}
		case "candidateAdapter":
			server.providerCandidates[step.Generation] = connections[step.Adapter]
		case "invoke":
			manifest := provider.InstallManifest{Provider: provider.ProviderRef{ID: step.Provider}}
			value := server.newProviderOperationCapability(manifest, provider.ServiceDefinition{ID: step.Service}, step.Operation, step.Generation, step.Candidate)
			response, failure := value.Invoke(context.Background(), nil, tools.ExecutionBinding{}, nil)
			result["toolError"] = response != nil && response.IsError
			if failure != nil {
				err = failure
			} else if response != nil && response.IsError {
				text := response.Content[0].(*mcp.TextContent).Text
				result["invokeError"] = strings.TrimPrefix(text, "Error: ")
			}
			if err != nil {
				result["invokeError"] = err.Error()
				err = nil
			}
			if _, ok := result["invokeError"]; !ok {
				result["invokeError"] = ""
			}
			sessions := map[string]int{}
			for _, id := range []string{step.Generation, "g1", "g2"} {
				if observed, ok := server.providerLifecycle.Get(id); ok {
					sessions[id] = observed.Sessions
				}
			}
			result["sessions"] = sessions
		case "restore":
			err = server.providerLifecycle.Restore(step.Restore)
		case "ready":
			err = server.providerLifecycle.MarkReady(step.Generation)
		case "activate":
			_, _, err = server.providerLifecycle.Activate(step.Generation)
		case "mount":
			err = server.mountProviderGeneration(step.Manifest, step.Generation, connections[step.Adapter])
		case "unmount":
			err = server.unmountProviderGeneration(step.Provider, step.Generation)
		case "ids":
			err = server.unmountProviderPlugins(step.IDs)
		case "displace":
			err = server.displaceCordisCapabilityFamilies(step.Provider, step.Families)
		case "close":
			err = server.providerContext.Dispose(context.Background())
		case "lookup":
			value, found := server.providerServiceValueFor(step.Provider, step.Service, step.Generation)
			result["found"] = found
			if found {
				result["service"] = map[string]any{"key": value.key, "provider": value.providerID, "generation": value.generationID, "capability": value.capabilityID, "offering": value.OfferingKey(), "inject": value.inject}
			}
			result["generationAdapter"] = server.providerGenerationAdapter(step.Provider, step.Generation) != nil
		case "families":
		default:
			panic("unsupported mount fixture action: " + step.Action)
		}
		message := ""
		if err != nil {
			message = err.Error()
		}
		result["error"] = message
		result["plugins"] = server.providerContext.PluginIDs()
		result["services"] = server.providerContext.ServiceKeys()
		result["families"] = server.injectableServiceFamilies(step.Generation)
		probes := map[string]string{}
		for id, connection := range connections {
			_, failure := connection.Call(context.Background(), "probe", nil)
			if failure == nil {
				panic("unconnected adapter unexpectedly invoked")
			}
			probes[id] = failure.Error()
		}
		result["adapters"] = probes
		observations = append(observations, result)
	}
	if err := server.providerContext.Dispose(context.Background()); err != nil {
		panic(err)
	}
	return observations
}
