// Parity-only bridge to the owning native schema compilation boundary.
package catalog

import (
	"encoding/json"
	"fmt"
	"github.com/google/jsonschema-go/jsonschema"
	"math"
	"reflect"
)

func ParityCheckSchema(label string, value map[string]any) error {
	return validateJSONSchema(label, value)
}

func ParitySchemaResolver(value map[string]any, action, pointer string) map[string]any {
	raw, err := json.Marshal(value)
	if err != nil {
		panic(err)
	}
	var schema jsonschema.Schema
	if err := json.Unmarshal(raw, &schema); err != nil {
		return map[string]any{"error": "fixture schema is malformed: " + err.Error()}
	}
	if action == "structure" {
		paths, err := jsonschema.ParityStructure(&schema)
		message := ""
		if err != nil {
			message = err.Error()
		}
		return map[string]any{"error": message, "paths": paths}
	}
	if action != "pointer" {
		panic("unknown schema resolver fixture action")
	}
	target, err := jsonschema.ParityDereference(&schema, pointer)
	message := ""
	if err != nil {
		message = err.Error()
	}
	return map[string]any{"error": message, "nil": target == nil, "schema": paritySchemaProjection(reflect.ValueOf(target))}
}

// Reflect the actual decoded typed tree, including nil vs empty collections
// and raw/const presence. This does not call Resolve or claim compiler parity.
func ParityDecodeSchema(value map[string]any) map[string]any {
	raw, err := json.Marshal(value)
	if err != nil {
		panic(err)
	}
	var schema jsonschema.Schema
	if err := json.Unmarshal(raw, &schema); err != nil {
		return map[string]any{"error": "fixture schema is malformed: " + err.Error(), "schema": nil}
	}
	return map[string]any{"error": "", "schema": paritySchemaProjection(reflect.ValueOf(schema))}
}

func paritySchemaProjection(v reflect.Value) any {
	if !v.IsValid() {
		return nil
	}
	if v.Type() == reflect.TypeFor[json.RawMessage]() {
		var value any
		if !v.IsNil() {
			if err := json.Unmarshal(v.Bytes(), &value); err != nil {
				panic(err)
			}
		}
		return map[string]any{"present": !v.IsNil(), "value": paritySchemaProjection(reflect.ValueOf(value)), "raw": string(v.Bytes())}
	}
	switch v.Kind() {
	case reflect.Pointer, reflect.Interface:
		if v.IsNil() {
			return nil
		}
		return paritySchemaProjection(v.Elem())
	case reflect.Struct:
		out := map[string]any{}
		for i := range v.NumField() {
			field := v.Type().Field(i)
			if !field.IsExported() {
				continue
			}
			item := v.Field(i)
			if field.Name == "Const" {
				out[field.Name] = map[string]any{"present": !item.IsNil(), "value": paritySchemaProjection(item)}
			} else {
				out[field.Name] = paritySchemaProjection(item)
			}
		}
		return out
	case reflect.Map:
		if v.IsNil() {
			return nil
		}
		out := map[string]any{}
		for _, key := range v.MapKeys() {
			out[key.String()] = paritySchemaProjection(v.MapIndex(key))
		}
		return out
	case reflect.Slice:
		if v.IsNil() {
			return nil
		}
		out := make([]any, v.Len())
		for i := range v.Len() {
			out[i] = paritySchemaProjection(v.Index(i))
		}
		return out
	case reflect.Float64:
		return map[string]any{"$float64": fmt.Sprintf("%016x", math.Float64bits(v.Float())), "value": v.Float()}
	default:
		return v.Interface()
	}
}

// Read the decoded integer as well as its refusal; successful cases must
// preserve the value, not merely return no error.
func ParitySchemaInteger(value map[string]any) *int {
	raw, err := json.Marshal(value)
	if err != nil {
		panic(err)
	}
	var schema jsonschema.Schema
	if err := json.Unmarshal(raw, &schema); err != nil {
		panic(err)
	}
	for key := range value {
		switch key {
		case "minLength":
			return schema.MinLength
		case "maxLength":
			return schema.MaxLength
		case "minItems":
			return schema.MinItems
		case "maxItems":
			return schema.MaxItems
		case "minProperties":
			return schema.MinProperties
		case "maxProperties":
			return schema.MaxProperties
		case "minContains":
			return schema.MinContains
		case "maxContains":
			return schema.MaxContains
		default:
			panic("integer fixture contains an unknown keyword")
		}
	}
	panic("integer fixture has no keyword")
}
